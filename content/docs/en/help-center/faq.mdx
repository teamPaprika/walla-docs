import { Accordion, Accordions } from 'fumadocs-ui/components/accordion';

<Accordions>
  <Accordion title="Is Walla free to use?">
    Yes, Walla offers a free plan. You can create forms, collect responses, and access core features at no cost. Check out our paid plans if you need more advanced features.
  </Accordion>

  <Accordion title="Where can I view response data?">
    Click the [Responses] tab in the form dashboard to view collected responses. You can also download the data in Excel or CSV format.
  </Accordion>

  <Accordion title="How do I share my form?">
    Click the [Share] button in the top right corner of the form editor to copy the link or share it via social media.
  </Accordion>

  <Accordion title="Can respondents attach files?">
    Yes, add a file upload question type to your form to allow respondents to attach images, documents, and other files.
  </Accordion>

  <Accordion title="Will response data be deleted if I delete a form?">
    Yes, deleting a form will also delete all associated response data. Make sure to download your response data before deleting the form.
  </Accordion>
</Accordions>
