import { Step, Steps } from 'fumadocs-ui/components/steps';

## Cargar una plantilla

<Steps>
  <Step>
    ### Pulsa el botón Crear proyecto
    En la pantalla del espacio de trabajo, haz clic en `Crear proyecto`.
  </Step>
  <Step>
    ### Selecciona Usar plantilla
    Elige `Usar plantilla`.
  </Step>
  <Step>
    ### Elige la plantilla que quieras
    Selecciona la plantilla de equipo que prefieras y pulsa `Usar plantilla`.
  </Step>
  <Step>
    ### Listo
    Se crea un proyecto nuevo a partir de la plantilla elegida.
  </Step>
</Steps>
