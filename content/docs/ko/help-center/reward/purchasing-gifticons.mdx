import { Step, Steps } from 'fumadocs-ui/components/steps';

## 리워드 구매 방법

<Steps>
  <Step>
    ### 리워드 관리 페이지로 이동
    `리워드 관리` 페이지로 이동합니다.

    <img src="/images/guides/reward/step-01-reward-management.png" alt="리워드 관리 페이지" style={{ width: "100%", borderRadius: "8px", marginTop: "12px", border: "1px solid var(--border)"}} />
  </Step>
  <Step>
    ### 리워드 구매 클릭
    상단의 `리워드 구매` 버튼을 클릭합니다.

    <img src="/images/guides/reward/step-02-purchase-button.png" alt="리워드 구매 버튼" style={{ width: "100%", borderRadius: "8px", marginTop: "12px", border: "1px solid var(--border)"}} />
  </Step>
  <Step>
    ### 리워드 선택
    제공되는 브랜드 리스트(예: 스타벅스, 배달의민족 등) 중 원하는 리워드를 선택합니다.

    <img src="/images/guides/reward/step-03-reward-select.png" alt="리워드 선택" style={{ width: "100%", borderRadius: "8px", marginTop: "12px", border: "1px solid var(--border)"}} />
  </Step>
  <Step>
    ### 수량 선택 및 결제
    리워드 종류와 수량을 선택하고 결제를 진행합니다.

    <img src="/images/guides/reward/step-04-payment.png" alt="결제 확인" style={{ width: "100%", borderRadius: "8px", marginTop: "12px", border: "1px solid var(--border)"}} />

    <img src="/images/guides/reward/step-04-payment-card.png" alt="결제 정보 입력" style={{ width: "100%", borderRadius: "8px", marginTop: "12px", border: "1px solid var(--border)"}} />

    <Callout type="warning" title="주의사항">
    - 왈라는 해외 PG사(Payment Gateway) 를 사용하고 있어, 카드사 정책에 따라 해외 결제 수수료가 부과될 수 있습니다.
    - 결제 전 반드시 카드사 앱이나 홈페이지에서 '해외 원화 결제 차단'(DCC 차단) 설정을 해제해주세요. (해당 설정이 활성화되어 있으면 결제가 진행되지 않아요.)
    </Callout>
  </Step>
  <Step>
    ### 구매 완료
    결제가 완료되면 리워드 구매가 완료되었다는 알림이 표시됩니다.

    <img src="/images/guides/reward/step-05-purchase-complete.png" alt="구매 완료" style={{ width: "100%", borderRadius: "8px", marginTop: "12px", border: "1px solid var(--border)"}} />
  </Step>
  <Step>
    ### 발송 진행
    리워드 관리 페이지에서 구매 내역을 확인하거나 응답자에게 리워드 발송을 진행합니다.
  </Step>
</Steps>
