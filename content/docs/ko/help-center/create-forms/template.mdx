import { Step, Steps } from 'fumadocs-ui/components/steps';

## 템플릿 불러오기

<Steps>
  <Step>
    ### 프로젝트 만들기 버튼 클릭
    워크스페이스 화면에서 `프로젝트 만들기` 버튼을 클릭합니다.

    <img src="/images/guides/manage-forms/team-template-create-project.png" alt="워크스페이스 우측 상단의 프로젝트 만들기 버튼" style={{ width: "100%", borderRadius: "8px", marginTop: "12px", marginBottom: "12px", border: "1px solid var(--border)" }} />
  </Step>
  <Step>
    ### 템플릿 사용하기 선택
    `템플릿 사용하기`를 선택합니다.

    <img src="/images/guides/manage-forms/team-template-select-template.png" alt="새로운 프로젝트 만들기 팝업의 템플릿 사용하기 선택" style={{ width: "100%", borderRadius: "8px", marginTop: "12px", marginBottom: "12px", border: "1px solid var(--border)" }} />
  </Step>
  <Step>
    ### 원하는 템플릿 선택
    원하는 팀 템플릿을 선택한 후 `템플릿 사용하기`를 클릭합니다.

    <img src="/images/guides/manage-forms/template-preview-use.png" alt="템플릿 미리보기 화면의 템플릿 사용하기 버튼" style={{ width: "100%", borderRadius: "8px", marginTop: "12px", marginBottom: "12px", border: "1px solid var(--border)" }} />
  </Step>
  <Step>
    ### 완료
    선택한 템플릿을 기반으로 새 프로젝트가 생성됩니다.

    <img src="/images/guides/manage-forms/template-project-created.png" alt="템플릿을 기반으로 생성된 새 프로젝트 편집 화면" style={{ width: "100%", borderRadius: "8px", marginTop: "12px", marginBottom: "12px", border: "1px solid var(--border)" }} />
  </Step>
</Steps>
